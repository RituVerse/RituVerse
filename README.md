```cpp
// Kadane's Algorithm — maximum subarray sum
// O(n) time · O(1) space
int maxSubArray(vector<int>& nums) {
    int best = nums[0];
    int curr = nums[0];
    for (int i = 1; i < nums.size(); ++i) {
        curr = max(nums[i], curr + nums[i]);
        best = max(best, curr);
    }
    return best;
}
```
